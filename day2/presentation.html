sealed class: كل الحالات معروفة — والمترجم يتأكد أنك غطيتها

  • كل الأنواع الفرعية في نفس الملف ← لا أحد يضيف حالة من الخارج.
  • switch expression بدون default: التغطية الكاملة مضمونة.
  • أضفت حالة ونسيت معالجتها؟ خطأ وقت الترجمة وليس crash عند المستخدم.
  • Circle(:final radius) = نمط كائن: يفحص النوع ويستخرج الحقل معاً.
جرّب: احذف سطر Square ← خطأ ترجمة: not exhaustively matched
sealed class Shape {}

class Circle extends Shape {
  Circle(this.radius);
  final double radius;
}

class Square extends Shape {
  Square(this.side);
  final double side;
}
// ...
// switch expression: لا نحتاج default لأن الـ sealed تضمن التغطية الكاملة
double area(Shape shape) => switch (shape) {
  Circle(:final radius) => 3.14 * radius * radius,
  Square(:final side) => side * side,
  Rectangle(:final width, :final height) => width * height,
};

String? lesson5Guard({required bool loggedIn, required Uri uri}) {
  final path = uri.path; // بدون day2://open — المسار فقط
  final isProtected = path.startsWith('/l5/account');
  final isLogin = path == '/l5/login';
  if (!loggedIn && isProtected) {
    return '/l5/login?from=${Uri.encodeComponent(path)}';
  }
  if (loggedIn && isLogin) {
    return uri.queryParameters['from'] ?? '/l5';
  }
  return null; // 3) لا تغيير
}

Dio واحد للتطبيق عبر Provider

  • BaseOptions: رابط أساسي + مهلات + headers مشتركة.
  • الترتيب مهم: HeadersInterceptor أولاً ليرى المسجّل الـ headers المضافة.

  • ref.onDispose(dio.close) يغلق الاتصالات.
final dioProvider = Provider<Dio>((ref) {
  final dio = Dio(
    BaseOptions(
      baseUrl: 'https://dummyjson.com',
      connectTimeout: const Duration(seconds: 10),
      receiveTimeout: const Duration(seconds: 10),
      headers: {'Accept': 'application/json'},
    ),
  );

  dio.interceptors.addAll([
    HeadersInterceptor(),
    ConsoleLogInterceptor(ref.read(networkLogProvider.notifier).add),
  ]);

  ref.onDispose(dio.close); // إغلاق الاتصالات عند حذف الـ provider
  return dio;
});

DioException ← sealed AppFailure — التحويل في مكان واحد

  • 1) AppFailure: sealed class (الدرس 1) بحالات يفهمها التطبيق.
  • 2) Repository: الطبقة الوحيدة التي تعرف Dio.
  • 3) الشاشة: AsyncValue + switch شامل.
  • switch على enum شامل: نوع جديد في Dio (مثل transformTimeout) ← خطأ ترجمة هنا.
factory AppFailure.fromDio(DioException e) {
  return switch (e.type) {
    DioExceptionType.connectionTimeout ||
    DioExceptionType.sendTimeout ||
    DioExceptionType.receiveTimeout => const TimeoutFailure(),
    DioExceptionType.connectionError => const NetworkFailure(),
    DioExceptionType.cancel => const CancelledFailure(),
    DioExceptionType.badResponse => switch (e.response?.statusCode) {
      401 || 403 => const UnauthorizedFailure(),
      404 => const NotFoundFailure(),
      final int code when code >= 500 => ServerFailure(code),
      final code => ServerFailure(code ?? 0, 'طلب غير صالح ($code)'),
    },
    DioExceptionType.badCertificate ||
    DioExceptionType.transformTimeout ||
    DioExceptionType.unknown => UnknownFailure(e.message ?? '$e'),
  };
}

  • 7 · الأخطاء: DioException ← AppFailure في Repository ← AsyncValue + switch.
