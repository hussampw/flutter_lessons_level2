Future = قيمة واحدة لاحقاً · Stream = قيم تصل مع الوقت

  • أمثلة: رسائل شات، موقع GPS، تقدّم تحميل… ونتائج قاعدة البيانات كلما تغيّرت (الدرس 6).
  • async* + yield لإنشاء Stream، و await for لاستهلاكه.
  • التحويلات تشبه دوال القوائم لكنها تعمل على قيم تصل مع الوقت: distinct · where · map · take.
Stream<int> countdown(int from) async* {
  for (var i = from; i >= 0; i--) {
    await Future.delayed(const Duration(milliseconds: 200));
    yield i; // yield ترسل قيمة إلى من يستمع
  }
}

  final _controller =
      StreamController<String>.broadcast();

  // نكشف الـ Stream فقط (للقراءة)
  Stream<String> get messages => _controller.stream;

  void send(String text) {
    if (_controller.isClosed) return;
    _controller.add(text); // إرسال قيمة
  }

  Future<void> dispose() => _controller.close();
}

class JsonFileStore {
  JsonFileStore(this.file);
  final File file;

  Future<List<String>> read() async {
    if (!await file.exists()) return []; // أول تشغيل
    final text = await file.readAsString();
    return (jsonDecode(text) as List).cast<String>();
  }

  Future<void> write(List<String> items) async {
    // نكتب في ملف مؤقت ثم نعيد تسميته
    final tmp = File('${file.path}.tmp');
    await tmp.writeAsString(jsonEncode(items), flush: true);
    await tmp.rename(file.path);
  }
}

  final file = File(p.join((await _dir()).path, fileName));
  if (await file.exists()) return (file, true); // ✅ بدون شبكة

  final res = await dio.get<List<int>>(
    url,
    options: Options(responseType: ResponseType.bytes),
  );
  await file.writeAsBytes(res.data!, flush: true);
  return (file, false);
}

// SELECT * FROM products WHERE price > ? ORDER BY price DESC
Future<List<Product>> productsAbove(double minPrice) =>
    (select(products)
          ..where((p) => p.price.isBiggerThanValue(minPrice))
          ..orderBy([(p) => OrderingTerm.desc(p.price)]))
        .get();

// UPDATE products SET price = price * 0.9 WHERE category_id = ?
Future<int> discountCategory(int categoryId) =>
    (update(products)..where((p) => p.categoryId.equals(categoryId))).write(
      ProductsCompanion.custom(price: products.price * const Constant(0.9)),
    );

// DELETE FROM products WHERE price < ?
Future<int> deleteCheaperThan(double price) =>
    (delete(products)..where((p) => p.price.isSmallerThanValue(price))).go();

stale-while-revalidate: اعرض المحفوظ فوراً، وحدّث في الخلفية

  1. اعرض الكاش من Drift فوراً (حتى لو قديماً = stale)
  2. انتهت صلاحيته (TTL)؟ اطلب من الـ API في الخلفية
  3. احفظ النتيجة (upsert) ← watch() يحدّث الواجهة وحده
  4. فشلت الشبكة؟ تبقى البيانات القديمة + رسالة ودّية
القاعدة: قاعدة البيانات هي «مصدر الحقيقة» للواجهة، والشبكة فقط تحدّثها.
بدون كاش: شاشة فارغة حتى يرد الخادم⏳ ≈ 800ms
مع SWR: القائمة من Drift فوراً≈ 20ms
بدون إنترنت: بدون كاش❌ شاشة خطأ
بدون إنترنت: مع SWR✅ آخر نسخة
💡 جرّب: افتح الشاشة ← فعّل «محاكاة عدم الاتصال» ← اخرج وادخل: القائمة ما زالت هنا.

refresh() يكتب في Drift، و revalidate() يقرر متى

  • isStale(): لا يوجد وقت تحديث، أو مرّ أكثر من ttl.
  • upsert للمنتجات + وقت التحديث في transaction واحدة.
  • الشبكة فشلت؟ حالة failed برسالة ودّية — والقائمة تبقى لأنها من Drift.
  • السحب للأسفل = revalidate(force: true).
try {
  if (!force && !await cache.isStale()) {
    // ✅ الكاش ما زال صالحاً ← لا طلب شبكة إطلاقاً
    emit(CacheStatus(CachePhase.fresh,
        fetchedAt: await cache.lastFetch()));
    return;
  }
  await cache.refresh();
} on DioException catch (e) {
  // ❌ الشبكة فشلت — البيانات القديمة تبقى معروضة
  emit(CacheStatus(
    CachePhase.failed,
    fetchedAt: await cache.lastFetch(),
    error: friendlyNetworkError(e),
  ));
}

for (final change in pending) {
  final task = Task.fromJson(jsonDecode(change.payload)
      as Map<String, dynamic>);
  final newer = await server.push(change.op, task);
  await db.transaction(() async {
    if (newer != null) {

  });
  pushed++;
}
